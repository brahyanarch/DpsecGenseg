     EXEC GEN_ObtenerMenuExpediente_SP 2193095, 'individual', 2 
     EXEC GEN_ObtenerMenuExpediente_SP 6202, 'grupal'  
     EXEC GEN_ObtenerMenuExpediente_SP 7379, 'grupal'

CREATE PROCEDURE [dbo].[GEN_ObtenerMenuExpediente_SP]
    @idSolicitud INT,
    @cTipoSolicitud VARCHAR(20),
    @idTipoFirma INT = -1
AS
BEGIN
    SET NOCOUNT ON;
    
    -- SECCIÓN 1: Declaración de Variables
    DECLARE @dFechaProduccion DATE = '2018-09-18';
    DECLARE @idTipoFirmaElectronica INT;
    DECLARE @idCuenta INT;
    DECLARE @nMontoDesembolso DECIMAL;
    DECLARE @idGrupoSolidario INT;
    DECLARE @idSolicitudOrigen INT;
    DECLARE @lRetencionAprobado BIT = 0;
    DECLARE @dRetencionAprobado DATE;
    DECLARE @cDescTipoDoc VARCHAR(120);
    DECLARE @cExpediente VARCHAR(120);
    DECLARE @idDescPosIntgr INT = 0;
    DECLARE @dFechaSis DATE;
    DECLARE @dFecRptPosIntConsolNuevo DATE;
    DECLARE @idDescPosConsCliente INT = 0;
    DECLARE @idDescDNI INT = 0;
    DECLARE @lDevuelto INT = 0;
    DECLARE @idDescInfoGral INT;
    DECLARE @idSentinel INT;
    DECLARE @cMenuTasa VARCHAR(250);
    DECLARE @idOpeTasa INT;
    DECLARE @FechaVigencia VARCHAR(100);
    DECLARE @idDescTipoDocInte INT;
    DECLARE @idDescTipoDocGrup INT;
    DECLARE @idArchivoFinal INT;
    DECLARE @idTipoArchivoSBS INT;
    DECLARE @_idDescTipoDoc INT;
    DECLARE @_idOperacion INT;
    DECLARE @_idTipoSolTasaDesembolso INT;
    DECLARE @_idOperacionSOL INT;
    
    -- OPTIMIZACIÓN: Consolidar consultas a SI_FinSolicitud
    DECLARE @dFechaRegistro DATE, @dFechaDecisFinal DATE, @idModoDesembolsoCli INT;
    
    SELECT @dFechaRegistro = dFechaRegistro,
           @dFechaDecisFinal = dFechaDecisFinal,
           @idModoDesembolsoCli = idModoDesembolsoCli,
           @_idOperacionSOL = idOperacion
    FROM dbo.SI_FinSolicitud
    WHERE idSolicitud = @idSolicitud;

    -- Inicializar idTipoFirma
    IF (@idTipoFirma <= 0)
    BEGIN
        SELECT @idTipoFirma = idTipoFirma
        FROM dbo.SI_FinSolicitud
        WHERE idSolicitud = @idSolicitud;
    END;
    
    SELECT @idTipoFirmaElectronica = idTipoFirma
    FROM dbo.SI_FinTipoFirma
    WHERE cTipoFirma = 'FIRMA ELECTRÓNICA KEYNUA';

    -- SECCIÓN 2: Preparar Condonaciones
    DECLARE @tmpCondonacion TABLE
    (
        idMenu INT,
        cMenu VARCHAR(120),
        idPadre INT,
        idDescTipoDoc INT,
        nOrden INT,
        idCategoriaArchivo INT,
        idArchivo INT,
        cExtension VARCHAR(120),
        idSecundario INT
    );
    
    SELECT @idCuenta = idCuenta
    FROM dbo.SI_FinCreditos
    WHERE idSolicitud = @idSolicitud
          AND @cTipoSolicitud = 'individual';

    IF EXISTS
    (
        SELECT 1
        FROM SI_FinSolicitudAprobacion
        WHERE idDocumento = @idCuenta
              AND idEstadoAproba NOT IN ( 4, 7 )
              AND lVigente = 1
    )
    BEGIN

        INSERT INTO @tmpCondonacion
        (
            idMenu,
            cMenu, 
            idPadre, 
            idDescTipoDoc, 
            nOrden,
            idCategoriaArchivo, 
            idArchivo, 
            cExtension, 
            idSecundario
        )
        SELECT idMenu = C.idDescTipoDoc * 100 + A.idTipoDoc + 1,
               cMenu = F.cDocumento,
               idPadre = D.idSolicitudAproba,
               C.idDescTipoDoc,
               nOrden = C.nOrden + (DATEDIFF(DAY, D.dFechaReg, GETDATE()))
                        + ROW_NUMBER() OVER (PARTITION BY C.idDescTipoDoc ORDER BY D.idSolicitudAproba ASC),
               C.idCategoriaArchivo,
               idArchivo = A.idTipoDoc,
               cExtension = '.pdf',
               idSecundario = D.idSolicitudAproba
        FROM dbo.SI_FINSolDocumentosCondonacion A
            INNER JOIN dbo.SI_FinCreditos B 
                ON A.idCuenta = B.idCuenta
            INNER JOIN dbo.SI_FinSolicitudAprobacion D 
                ON A.idCuenta = D.idDocumento
                   AND A.idSolApr = D.idSolicitudAproba
                   AND D.idEstadoAproba NOT IN ( 4, 7 )
            INNER JOIN dbo.SI_FinEstadoAproba E 
                ON E.idEstadoAproba = D.idEstadoAproba
            INNER JOIN dbo.SI_FINTipoDocCondonacion F 
                ON F.idTipoDoc = A.idTipoDoc
            CROSS JOIN dbo.SI_FINDescTipoDoc C
        WHERE B.idSolicitud = @idSolicitud
              AND A.cNombreDoc <> ''
              AND A.lVigente = 1
              AND C.cDescTipoDoc = 'SOLICITUD DE CONDONACIÓN'
              AND C.lVigente = 1
              AND @cTipoSolicitud = 'individual';

        INSERT INTO @tmpCondonacion
        (
            idMenu, 
            cMenu, 
            idPadre, 
            idDescTipoDoc, 
            nOrden,
            idCategoriaArchivo, 
            idArchivo, 
            cExtension, 
            idSecundario
        )
        SELECT idMenu = MIN(idMenu) - 1,
                cMenu = 'Informe - Propuesta de Negociación',
                idPadre, 
                idDescTipoDoc,
                nOrden = MIN(nOrden),
                idCategoriaArchivo,
                idArchivo = 0,
                cExtension = '.rdlc',
                idSecundario
        FROM @tmpCondonacion
        GROUP BY idPadre, 
                idDescTipoDoc, 
                idCategoriaArchivo, 
                idSecundario
        UNION ALL
        SELECT idMenu = D.idSolicitudAproba,
               cMenu = CONCAT(
                   'Solicitud: ',
                   D.idSolicitudAproba,
                   ' (',
                   CONVERT(VARCHAR(10), D.dFechaReg, 103),
                   ') ',
                   E.cEstadoAproba
               ),
               idPadre = C.idDescTipoDoc,
               C.idDescTipoDoc,
               nOrden = C.nOrden + (DATEDIFF(DAY, D.dFechaReg, GETDATE())),
               idCategoriaArchivo = 0,
               idArchivo = 0,
               cExtension = '.pdf',
               idSecundario = 0
        FROM dbo.SI_FINSolDocumentosCondonacion A
            INNER JOIN dbo.SI_FinCreditos B 
                ON A.idCuenta = B.idCuenta
            INNER JOIN dbo.SI_FinSolicitudAprobacion D 
                ON A.idCuenta = D.idDocumento
                   AND A.idSolApr = D.idSolicitudAproba
                   AND D.idEstadoAproba NOT IN ( 4, 7 )
            INNER JOIN dbo.SI_FinEstadoAproba E 
                ON E.idEstadoAproba = D.idEstadoAproba
            INNER JOIN dbo.SI_FINTipoDocCondonacion F 
                ON F.idTipoDoc = A.idTipoDoc
            CROSS JOIN dbo.SI_FINDescTipoDoc C
        WHERE B.idSolicitud = @idSolicitud
              AND A.cNombreDoc <> ''
              AND A.lVigente = 1
              AND C.cDescTipoDoc = 'SOLICITUD DE CONDONACIÓN'
              AND C.lVigente = 1
              AND @cTipoSolicitud = 'individual'
        GROUP BY D.idSolicitudAproba, 
            C.idDescTipoDoc, 
            E.cEstadoAproba, 
            C.nOrden, 
            D.dFechaReg;
    END;

    -- SECCIÓN 3: Cargar Archivos
    DECLARE @tmpArchivos TABLE
    (
        idArchivos INT,
        cNombreArchivo VARCHAR(360),
        idTipoArchivo INT,
        idCli INT,
        idSolicitud INT,
        idCategoriaArchivo INT,
        idDescTipoDoc INT,
        idSolicitudCredGrupoSol INT,
        idArchivoForeing INT,
        idFormato INT,
        dFechaReg DATE,
        lVigente BIT,
        dFechaMod DATE,
        idModulo INT,
        cDetalleArchivo VARCHAR(200)
    );

    INSERT INTO @tmpArchivos
    (
        idArchivos      , cNombreArchivo        , idTipoArchivo     , idCli                     , 
        idSolicitud     , idCategoriaArchivo    , idDescTipoDoc     , idSolicitudCredGrupoSol   ,
        idArchivoForeing, idFormato             , dFechaReg         , lVigente                  , 
        dFechaMod       , idModulo              , cDetalleArchivo
    )
    SELECT idArchivos, 
        cNombreArchivo, 
        idTipoArchivo, 
        idCli, 
        idSolicitud,
        idCategoriaArchivo, 
        idDescTipoDoc, 
        idSolicitudCredGrupoSol,
        idArchivoForeing, 
        idFormato, 
        dFechaReg, 
        lVigente, 
        dFechaMod,
        idModulo, 
        cDetalleArchivo
    FROM dbo.SI_FinArchivo
    WHERE (
              @cTipoSolicitud = 'individual'
              AND idSolicitud = @idSolicitud
          )
          OR (
                @cTipoSolicitud = 'grupal'
                AND idSolicitudCredGrupoSol = @idSolicitud
            )
            AND
            (
                idDescTipoDoc = 19
                AND lVigente = 1
                OR idDescTipoDoc <> 19
            );

    -- SECCIÓN 4: Solicitudes ORIGEN TasaPost
    SET @idSolicitudOrigen = 0;
    
    SELECT @idSolicitudOrigen = idSolicitudOrigen
    FROM SI_FINSoliCambioCredito
    WHERE idSolicitud = @idSolicitud
          AND idOperacion = 7;

    IF EXISTS
    (
        SELECT 1
        FROM SI_FinTasaNegociada
        WHERE idSolicitud = @idSolicitudOrigen
              AND lVigente = 1
              AND ISNULL(idTipoSolTasaDesembolso, 1) = 2
    )
    BEGIN
        IF EXISTS
        (
            SELECT 1
            FROM SI_FINSoliCambioCredito
            WHERE idSolicitud = @idSolicitud
                  AND idOperacion = 7
        )
        BEGIN
            -- OPTIMIZACIÓN: Inserción directa sin tabla temporal #FilteredArchivos1
            INSERT INTO @tmpArchivos
            (
                idArchivos          , cNombreArchivo    , idTipoArchivo     , idCli                     , 
                idSolicitud         , idCategoriaArchivo, idDescTipoDoc     , idSolicitudCredGrupoSol   ,
                idArchivoForeing    , idFormato         , dFechaReg         , lVigente                  , 
                dFechaMod           , idModulo          , cDetalleArchivo
            )
            SELECT idArchivos,
                cNombreArchivo, 
                idTipoArchivo, 
                idCli, 
                idSolicitud,
                idCategoriaArchivo, 
                idDescTipoDoc, 
                idSolicitudCredGrupoSol,
                idArchivoForeing, 
                idFormato, 
                dFechaReg, 
                lVigente, 
                dFechaMod,
                idModulo, 
                cDetalleArchivo
            FROM dbo.SI_FinArchivo
            WHERE (
                      @cTipoSolicitud = 'individual'
                      AND idSolicitud = @idSolicitudOrigen
                  )
                  OR (
                         @cTipoSolicitud = 'grupal'
                         AND idSolicitudCredGrupoSol = @idSolicitudOrigen
                     )
                     AND
                     (
                         idDescTipoDoc = 19
                         AND lVigente = 1
                         OR idDescTipoDoc <> 19
                     );

            IF EXISTS
            (
                SELECT 1
                FROM SI_FINSoliCambioCredito
                WHERE idSolicitud = @idSolicitud
                      AND idOperacion = 7
                      AND ISNULL(nReprogramacion, 0) != 0
            )
            BEGIN
                SET @lRetencionAprobado = 1;
                SET @dRetencionAprobado =
                (
                    SELECT dFechaOpe
                    FROM SI_FINSoliCambioCredito
                    WHERE idSolicitud = @idSolicitud
                          AND idOperacion = 7
                );
            END;
        END;
    END;

    -- SECCIÓN 5: Tablas de trabajo para menú
    DECLARE @tmpMenu TABLE
    (
        idMenu INT,
        cMenu VARCHAR(500),
        idPadre INT,
        idDescTipoDoc INT,
        nOrden FLOAT,
        idCategoriaArchivo INT,
        idArchivo INT,
        cExtension VARCHAR(5),
        idSecundario INT
    );

    DECLARE @tmpDescTipoDoc TABLE
    (
        idExpediente INT,
        nOrden INT,
        idDetGrupo INT,
        idDescTipoDoc INT
    );

    INSERT INTO @tmpDescTipoDoc 
    (
        idExpediente, 
        nOrden, 
        idDetGrupo, 
        idDescTipoDoc
    )
    EXEC dbo.GEN_IdExpedientesGrupo_SP 0, @idSolicitud, @cTipoSolicitud, 1;

    -- SECCIÓN 6: Posición Integral Intervinientes
    SELECT @idDescPosIntgr = idDescTipoDoc
    FROM dbo.SI_FINDescTipoDoc
    WHERE cDescTipoDoc = 'Posición Integral de Intervinientes';

    IF (@dFechaRegistro >= @dFechaProduccion OR @dFechaDecisFinal >= @dFechaProduccion)
    BEGIN
        INSERT INTO @tmpMenu
        (
            idMenu, 
            cMenu, 
            idPadre, 
            idDescTipoDoc, 
            nOrden,
            idCategoriaArchivo, 
            idArchivo, 
            cExtension, 
            idSecundario
        )
        SELECT idMenu = -1,
               cDescMenu = CONCAT(
                   '(R) ',
                   AA.cTipoIntervencion,
                   ' (',
                   Cl.cDocumentoID,
                   ') ',
                   CASE 
                       WHEN D.dFechaReg IS NULL THEN ''
                       ELSE CONVERT(VARCHAR(10), D.dFechaReg, 103)
                   END
               ),
               idPadre = B.idDescTipoDoc,
               B.idDescTipoDoc,
               nOrden = C.nOrden + A.idTipoInterv + 1,
               B.idCategoriaArchivo,
               idArchivos = ISNULL(D.idArchivos, 0),
               cExtension = '.rdlc',
               idSecundario = A.idCli
        FROM dbo.SI_FinCliIntervCre A
            INNER JOIN dbo.SI_FinCliente Cl 
                ON Cl.idCli = A.idCli 
                AND Cl.lVigente = 1
            INNER JOIN dbo.SI_FinTipoIntervencionCre AA 
                ON AA.idTipoInterv = A.idTipoInterv
                   AND AA.idTipoInterv NOT IN ( 2, 4, 12 )
                   AND A.lEstado = 1
                   AND A.idSolicitud = @idSolicitud
                   AND A.idModulo = 1
            CROSS JOIN dbo.SI_FINDescTipoDoc B
            INNER JOIN dbo.SI_FINDescTipoDoc C 
                ON C.idDescTipoDoc = B.idPadre 
                AND C.lVigente = 1
            CROSS JOIN dbo.SI_FinExpediente E
            LEFT JOIN @tmpArchivos D
                ON D.idCategoriaArchivo = B.idCategoriaArchivo
                   AND D.idCli = A.idCli
                   AND D.lVigente = 1
                   AND D.idDescTipoDoc = @idDescPosIntgr
                   AND @idSolicitud = CASE WHEN @cTipoSolicitud = 'individual' 
                                           THEN D.idSolicitud 
                                           ELSE D.idSolicitudCredGrupoSol END
        WHERE E.cStoreProcedure = 'CRE_RptPosicionIntegralIntervinientes_SP'
              AND B.cDescTipoDoc = 'Posición Integral de Intervinientes'
              AND @cTipoSolicitud = 'individual';
    END;

    -- SECCIÓN 7: Posición Consolidada Cliente
    SELECT @dFechaSis = CAST(cValVar AS DATE) 
    FROM dbo.SI_FinVariable 
    WHERE cVariable = 'dFechaAct' AND idAgencia = 1;
    
    SELECT @dFecRptPosIntConsolNuevo = CAST(cValVar AS DATE) 
    FROM dbo.SI_FinVariable 
    WHERE cVariable = 'dFecRptPosIntConsolNuevo';
    
    SELECT @idDescPosConsCliente = idDescTipoDoc
    FROM dbo.SI_FINDescTipoDoc
    WHERE cDescTipoDoc = 'Posición consolidada Cliente';

    IF (@dFechaRegistro >= @dFechaProduccion OR @dFechaDecisFinal >= @dFechaProduccion)
    BEGIN
        INSERT INTO @tmpMenu
        (
            idMenu, 
            cMenu, 
            idPadre, 
            idDescTipoDoc, 
            nOrden,
            idCategoriaArchivo, 
            idArchivo, 
            cExtension, 
            idSecundario
        )
        SELECT idMenu = -1,
               cDescMenu = CONCAT(
                   '(R) ',
                   AA.cTipoIntervencion,
                   ' (',
                   Cl.cDocumentoID,
                   ') ',
                   CASE 
                       WHEN D.dFechaReg IS NULL THEN ''
                       ELSE CONVERT(VARCHAR(10), D.dFechaReg, 103)
                   END
               ),
               idPadre = B.idDescTipoDoc,
               B.idDescTipoDoc,
               nOrden = C.nOrden + A.idTipoInterv + 1,
               B.idCategoriaArchivo,
               idArchivos = ISNULL(D.idArchivos, 0),
               cExtension = '.rdlc',
               idSecundario = A.idCli
        FROM dbo.SI_FinCliIntervCre A
            INNER JOIN dbo.SI_FinCliente Cl 
                ON Cl.idCli = A.idCli 
                    AND Cl.lVigente = 1
            INNER JOIN dbo.SI_FinTipoIntervencionCre AA 
                ON AA.idTipoInterv = A.idTipoInterv
                   AND AA.idTipoInterv NOT IN ( 2, 4, 12 )
                   AND A.lEstado = 1
                   AND A.idSolicitud = @idSolicitud
                   AND A.idModulo = 1
            CROSS JOIN dbo.SI_FINDescTipoDoc B
            INNER JOIN dbo.SI_FINDescTipoDoc C 
                ON C.idDescTipoDoc = B.idPadre 
                    AND C.lVigente = 1
            CROSS JOIN dbo.SI_FinExpediente E
            LEFT JOIN @tmpArchivos D
                ON D.idCategoriaArchivo = B.idCategoriaArchivo
                   AND D.idCli = A.idCli
                   AND D.lVigente = 1
                   AND D.idDescTipoDoc = @idDescPosConsCliente
                   AND @idSolicitud = CASE WHEN @cTipoSolicitud = 'individual' 
                                           THEN D.idSolicitud 
                                           ELSE D.idSolicitudCredGrupoSol END
        WHERE E.cStoreProcedure = 'CRE_RptPosicionConsolidadaCliente_SP'
              AND B.cDescTipoDoc = 'Posición consolidada Cliente'
              AND E.cExpediente = CASE 
                  WHEN @dFechaSis >= @dFecRptPosIntConsolNuevo 
                  THEN 'Posición Consolidada de Cliente Nuevo'
                  ELSE 'Posición consolidada Cliente'
              END
              AND @cTipoSolicitud = 'individual';
    END;

    -- SECCIÓN 8: DNI Intervinientes
    SELECT @idDescDNI = idDescTipoDoc
    FROM dbo.SI_FINDescTipoDoc
    WHERE cDescTipoDoc = 'DNI de Intervinientes';

    INSERT INTO @tmpMenu
    (
        idMenu, 
        cMenu, 
        idPadre, 
        idDescTipoDoc, 
        nOrden,
        idCategoriaArchivo, 
        idArchivo, 
        cExtension, 
        idSecundario
    )
    SELECT idMenu = -1,
           cDescMenu = CONCAT('(R) ', AA.cTipoIntervencion, ' (', Cl.cDocumentoID, ') '),
           idPadre = B.idDescTipoDoc,
           B.idDescTipoDoc,
           nOrden = C.nOrden + A.idTipoInterv * 10,
           B.idCategoriaArchivo,
           idArchivos = 0,
           cExtension = '.rdlc',
           idSecundario = A.idCli
    FROM dbo.SI_FinCliIntervCre A
        INNER JOIN dbo.SI_FinCliente Cl 
            ON Cl.idCli = A.idCli 
            AND Cl.lVigente = 1
        INNER JOIN dbo.SI_FinTipoIntervencionCre AA 
            ON AA.idTipoInterv = A.idTipoInterv
               AND A.lEstado = 1
               AND A.idSolicitud = @idSolicitud
               AND A.idModulo = 1
        CROSS JOIN dbo.SI_FINDescTipoDoc B
        INNER JOIN dbo.SI_FINDescTipoDoc C 
            ON C.idDescTipoDoc = B.idPadre 
            AND C.lVigente = 1
        CROSS JOIN dbo.SI_FinExpediente E
    WHERE E.cStoreProcedure = 'CLI_RecuperaDatosDNISolicitud_SP'
          AND B.cDescTipoDoc = 'DNI de Intervinientes'
          AND @cTipoSolicitud = 'individual';

    -- SECCIÓN 9: Expedientes Generados
    IF (@cTipoSolicitud = 'individual')
    BEGIN
        SELECT @lDevuelto = CASE WHEN idEstadoEvalCred IN ( 8, 9 ) THEN 1 ELSE 0 END
        FROM dbo.SI_FinEvalCred
        WHERE idSolicitud = @idSolicitud
              AND lAlterna = 0;
    END;
    ELSE
    BEGIN
        SELECT @lDevuelto = CASE WHEN idEstadoEvalCred IN ( 8, 9 ) THEN 1 ELSE 0 END
        FROM dbo.SI_FinEvalCredGrupoSol
        WHERE idSolicitudCredGrupoSol = @idSolicitud
        GROUP BY idEstadoEvalCred;
    END;

    INSERT INTO @tmpMenu
    (
        idMenu, 
        cMenu, 
        idPadre, 
        idDescTipoDoc, 
        nOrden,
        idCategoriaArchivo, 
        idArchivo, 
        cExtension, 
        idSecundario
    )
    SELECT idMenu = -1,
           cDescMenu = CONCAT(
               '(R) ',
               CASE
                   WHEN B.idDescTipoDoc = 34 THEN 'Solicitud Crédito Grupal'
                   WHEN B.idDescTipoDoc = 35 THEN 'Evaluación Cualitativa'
                   WHEN B.idDescTipoDoc = 36 THEN 'Evaluación Cuantitativa y Excepciones'
                   WHEN B.idDescTipoDoc = 37 THEN 'Acta de Resolución de crédito grupal'

	@idCli		INT,
	@dFecIni	DATE,
	@dFecFin	DATE,
	@nMontoIni	DECIMAL(14,2),
	@nMontoMax	DECIMAL(14,2),
	@idMoneda	INT,
	@idAgencia	INT,
	@idPerfil	INT, 
	@idUsuario	INT,
	@nModo INT,
	@idCanalAproCred INT,
	@idEstadoCre INT = 9999
AS
SET NOCOUNT ON
BEGIN
	
	IF(@nModo=0)
	BEGIN
		IF(@idPerfil = 64)
		BEGIN
			SET @idAgencia = 0;
		END

		SELECT idEval = A.idEvalCred
			,A.idNivelAprobacion
			,A.idEstadoEvalCred
			,A.idEtapaEvalCred
			,C.idSolicitud
			,nMontoProp = ISNULL(A.nMonPropuesto,0)
			,C.idCli
			,B.cNombre
			,C.IdMoneda
			,F.cMoneda
			,idAsesor = H.idUsuario
			,cAsesor = I.cNombre
			,A.idEvalCred
			,D.cProducto
			,C.IdProducto

			,J.cCanalAproCred
			,K.cNomCorto
			,L.cNombreEstab
			,A.idTipEvalCred
			, nTieneVisita = CAST(IIF(N.idSolicitud IS NULL, 0, 1) AS INT)
			, cComentario = ISNULL(n.cObservacion, '')
			, cCanalRegistro = ISNULL(O.cCanalRegistro, 'Core Bank')
			, c.idOperacion
			, nCuotaAprox = IIF(c.nCuotaAprox IS NULL, 0, c.nCuotaAprox)
            , idClasificacionInterna = ISNULL(b.idClasifInterna, CAST(0 AS INT))
			, C.idEstado AS idEstSol
			, R.cEstado AS cEstSol

		FROM SI_FinEvalCred A
		INNER JOIN SI_FinCliente B
			ON A.idCli = B.idCli
		INNER JOIN SI_FinSolicitud C
			ON A.idSolicitud = C.idSolicitud
		INNER JOIN dbo.SI_FinProducto D
			ON D.IdProducto = C.idProducto
		INNER JOIN SI_FinMoneda F

CREATE PROCEDURE CRE_PI2_LisCredPreApro_SP
	@idTipoDocumento INT,
	@cNumDocumento VARCHAR(20)
AS
SET NOCOUNT ON
BEGIN
	SELECT idCli, nVinculado
	INTO #tList1
	FROM CRE_identifPIV2_FN(@idTipoDocumento, @cNumDocumento, 0)

	CREATE TABLE #BasePreAprobados
	(
		idCreditosPreAprobados INT NOT NULL,
		idCliente INT NOT NULL,
		idAgencia INT NOT NULL,
		nMonto DECIMAL(14,2) NOT NULL,
		cProducto VARCHAR(100) NOT NULL,
		nPlazoCapitalTrabajo INT NOT NULL,
		nPlazoActivoFijo INT NOT NULL,
		nTasa DECIMAL(14,4) NOT NULL,
		lVigente BIT NULL,
		cNombreCampanha VARCHAR(200) NULL,
		nMontoPreAprobado DECIMAL(14,2) NULL,
		nPlazoCapitalTrabajoPreAprobado INT NULL,
		nPlazoActivoFijoPreAprobado INT NULL,
		lRatuki BIT NULL,
		idGrupoCamp INT NULL,
		idAsesor INT NULL,
		idMoneda INT NULL,
		idClasifInternaCamp INT NULL,
		cDocumentoID VARCHAR(15) NULL
	);

	IF EXISTS (SELECT 1 FROM #tList1)
	BEGIN
		INSERT INTO #BasePreAprobados
		SELECT A.*
		FROM [dbo].[SI_FinCreditosPreAprobados] A
		INNER JOIN #tList1 T ON A.idCliente = T.idCli
		WHERE A.lVigente = 1
	END
	ELSE
	BEGIN
		INSERT INTO #BasePreAprobados
		SELECT A.*
		FROM [dbo].[SI_FinCreditosPreAprobados] A
		WHERE A.cDocumentoID = @cNumDocumento
		  AND A.lVigente = 1
	END

	SELECT DISTINCT
		 A.idCreditosPreAprobados
		,A.idCliente
		,A.idAgencia
		,'0.00' AS nMonto
		,A.cProducto
		,A.nPlazoCapitalTrabajo
		,A.nPlazoActivoFijo
		,A1.nTasa
		,A.lVigente
		,A.cNombreCampanha
		,'Sujeto a evaluación' AS nMontoPreAprobado
		,A.nPlazoCapitalTrabajoPreAprobado
		,A.nPlazoActivoFijoPreAprobado
		,A.lRatuki
		,A.idGrupoCamp
		,A1.idAsesor
		,A.idMoneda
		,B.cNomCorto
		,C.cMoneda
		,E.cNombre AS cNomUser
		,A1.idDestino
		,F.cDestino
		,A1.nPlazo
		,T.nVinculado
	FROM #BasePreAprobados A
	LEFT JOIN #tList1 T ON A.idCliente = T.idCli
	LEFT JOIN SI_FinDetallePreAprobados A1 ON A.idCreditosPreAprobados = A1.idCreditosPreAprobados AND A1.lVigente = 1
	LEFT JOIN SI_FinAgencia B ON B.idAgencia = A.idAgencia
	LEFT JOIN SI_FinMoneda C ON C.idMoneda = A.idMoneda
	LEFT JOIN SI_FinUsuario D ON D.idUsuario = A1.idAsesor
	LEFT JOIN SI_FinCliente E ON E.idCli = D.idCli
	LEFT JOIN SI_FinDestinoCredito F ON A1.idDestino = F.idDestino

	DROP TABLE #tList1
	DROP TABLE #BasePreAprobados
END
GO

	/**/
	
	DECLARE @tmpParametros TABLE (
		cParametro VARCHAR(120) PRIMARY KEY,
		cValor VARCHAR(120)
	)
	INSERT INTO @tmpParametros
			( cParametro, cValor )
	VALUES  ('idSolicitud', @idSolicitud),
			('idCli', @idCli),
			('idEvalCred', @idEvalCred),
			('idSolicitudCredGrupoSol', @idSolicitud),
			('idEvalCredGrupoSol', @idEvalCredGrupoSol),
			('nFuente', 2),
			('idCliTitular', @idCli)
	/**/
	DECLARE @tmpExpedienteID TABLE (
		idExpediente INT, nOrden INT, idDetGrupo INT, idDescTipoDoc INT,
		INDEX IX_tmpExpedienteID (idExpediente)
	)
	 INSERT INTO @tmpExpedienteID
	         ( idExpediente, nOrden, idDetGrupo, idDescTipoDoc )
	EXEC GEN_IdExpedientesGrupo_SP @idGrupo, @idSolicitud, @cTipoSolicitud, 0
	/**/
	
	DECLARE @tmpParameters TABLE (
		nParameter INT, idParametro INT, idExpediente INT, cParametro VARCHAR(120), cValor VARCHAR(120),
		INDEX IX_tmpParameters (idExpediente, nParameter)
	)
	/*	GET Valores Parametros	*/
	INSERT INTO @tmpParameters
	        ( nParameter ,idParametro ,idExpediente ,cParametro ,cValor)
	SELECT 
		nParameter = ROW_NUMBER() OVER (PARTITION BY C.idExpediente ORDER BY C.idExpediente DESC), D.idParametro, C.idExpediente, D.cParametro, cValor = NULL
	FROM @tmpExpedienteID A
	INNER JOIN dbo.SI_FinExpediente C
		ON C.idExpediente = A.idExpediente
		AND C.lVigente = 1
	INNER JOIN dbo.SI_FinSpParametroExpediente D
		ON D.idExpediente = C.idExpediente
		AND D.lVigente = 1
	ORDER BY A.nOrden,D.nOrden
	
	UPDATE A
		SET A.cValor = B.cValor
	FROM @tmpParameters A
	INNER JOIN @tmpParametros B
		ON A.cParametro = B.cParametro

			IF EXISTS (SELECT 1 FROM @tmpParameters WHERE cValor IS NULL)
			BEGIN
				SET @cMsgRespuesta = 'No existe el valor de la variable ' + (SELECT TOP 1 cParametro FROM @tmpParameters WHERE cValor IS NULL)
			END
	
	DECLARE @tmpExpedientes TABLE (
		nExpediente INT PRIMARY KEY, idExpediente INT, cExpediente VARCHAR(120),
		cStoreProcedure VARCHAR(120), cReporte VARCHAR(120), lExiste BIT
	)

	/*	GET Expedientes	*/
	INSERT INTO @tmpExpedientes
	        ( nExpediente ,idExpediente ,cExpediente ,cStoreProcedure ,cReporte, lExiste)
	SELECT 
		nExpediente = ROW_NUMBER() OVER(ORDER BY C.idExpediente ASC), C.idExpediente, C.cExpediente, C.cStoreProcedure, C.cReporte,
		lExiste = CAST(0 AS BIT)
	FROM  @tmpExpedienteID A
		INNER JOIN dbo.SI_FinExpediente C
			ON A.idExpediente = C.idExpediente
			AND C.lVigente = 1
		INNER JOIN dbo.SI_FinDataSetExpediente D
			ON D.idExpediente = C.idExpediente
			AND D.lVigente = 1
	GROUP BY C.idExpediente, C.cExpediente,C.cReporte, C.cStoreProcedure, A.nOrden
	ORDER BY A.nOrden

	UPDATE T
		SET T.lExiste = CASE WHEN O.object_id IS NOT NULL THEN 1 ELSE 0 END
	FROM @tmpExpedientes T
	LEFT JOIN sys.objects O
		ON O.object_id = OBJECT_ID(N'[dbo].[' + T.cStoreProcedure + N']')
		AND O.type IN (N'P', N'PC')

	DECLARE @cStoreProcedure VARCHAR(120), @cSqlSP NVARCHAR(500)
	DECLARE @nExpedientes INT = 0, @nProc INT = 1, @idExpediente INT
	DECLARE @cParametros VARCHAR(120) = ''
	SELECT @nExpedientes = COUNT(idExpediente) FROM @tmpExpedientes 

	/*	Loop 1: ejecuta SOLO el primer SP del listado (nProc < 2), igual que el original	*/
	WHILE @nProc < 2
	BEGIN
		SELECT @cStoreProcedure = cStoreProcedure, @idExpediente = idExpediente, @cMsgRespuesta = IIF(lExiste = 0, 'No Existe el SP ' + cStoreProcedure, @cMsgRespuesta)
		FROM @tmpExpedientes WHERE nExpediente = @nProc

		IF NOT EXISTS (SELECT 1 FROM @tmpExpedientes WHERE nExpediente = @nProc AND lExiste = 1)
		BEGIN
			BREAK
		END

		SELECT @cParametros = STRING_AGG(cValor, ', ') WITHIN GROUP (ORDER BY nParameter)
		FROM @tmpParameters
		WHERE idExpediente = @idExpediente

		SET @cSqlSP = CONCAT('EXEC ',@cStoreProcedure,' ',@cParametros)
		EXEC sys.sp_executesql @cSqlSP
		
		SET @nProc = @nProc+1
	END

	DECLARE @lCongelarPosicion INT = 0
	IF(@cGrupo = 'Solicitud de Credito')
	BEGIN
	    IF EXISTS (SELECT 1 FROM dbo.SI_FinSolicitud WHERE idSolicitud = @idSolicitud AND idEstado = 2 AND idUsuApro IS NOT NULL AND idNivelAprobacion IS NOT NULL)
		BEGIN
		    SET @lCongelarPosicion = 1
		END
	END
	ELSE IF(@cGrupo = 'Aprobacion de Credito')
	BEGIN
	    SET @lCongelarPosicion = 1
	END

	IF(@lCongelarPosicion = 1)
	BEGIN
	    /*Posicion Integral*/
		DECLARE @tmpInterv TABLE (
			nNum INT PRIMARY KEY, idCli INT, inTipoInterv INT
		)
		INSERT INTO @tmpInterv
		        ( nNum, idCli, inTipoInterv )
		SELECT 
				nNum = ROW_NUMBER() OVER (ORDER BY A.idTipoInterv,A.idCli), A.idCli, A.idTipoInterv
			FROM dbo.SI_FinCliIntervCre A
			WHERE A.idTipoInterv NOT IN (2,4,12)
				AND A.lEstado = 1 
				AND A.idSolicitud = @idSolicitud
				AND A.idModulo = 1 
			ORDER BY A.idTipoInterv, A.idCli

			DECLARE @nNum INT = 1
			WHILE (SELECT COUNT(nNum) FROM @tmpInterv) >= @nNum
			BEGIN
				DECLARE @idCliTitular INT
				SELECT @idCliTitular = idCli FROM @tmpInterv WHERE nNum = @nNum
				EXEC dbo.CRE_RptPosicionIntegralIntervinientes_SP @idCliTitular
				SET @nNum = @nNum + 1
			END
	END
	
	/*	Loop 2: ejecuta el resto de expedientes, continuando desde donde quedó @nProc tras el Loop 1	*/
	WHILE @nProc <= @nExpedientes
	BEGIN
		SELECT @cStoreProcedure = cStoreProcedure, @idExpediente = idExpediente, @cMsgRespuesta = IIF(lExiste = 0, 'No Existe el SP ' + cStoreProcedure, @cMsgRespuesta)
		FROM @tmpExpedientes WHERE nExpediente = @nProc

		IF NOT EXISTS (SELECT 1 FROM @tmpExpedientes WHERE nExpediente = @nProc AND lExiste = 1)
		BEGIN
			BREAK
		END

		SELECT @cParametros = STRING_AGG(cValor, ', ') WITHIN GROUP (ORDER BY nParameter)
		FROM @tmpParameters
		WHERE idExpediente = @idExpediente

		SET @cSqlSP = CONCAT('EXEC ',@cStoreProcedure,' ',@cParametros)
		EXEC sys.sp_executesql @cSqlSP
		
		SET @nProc = @nProc+1
	END

	SELECT nRespuesta = 1 , cMsg='Ejecución completa'
END TRY
BEGIN CATCH
	DECLARE @cError VARCHAR(600) = ERROR_MESSAGE ()
	EXEC GEN_GuardarLogExpedientesLinea_SP @idSolicitud, @cGrupo, @cTipoSolicitud, @cError, @cSqlSP, @idUsuario
	SELECT nRespuesta = 0 , cMsg=CONCAT('Error: ',@cMsgRespuesta,' ',@cError)
END CATCH

GO

CREATE PROCEDURE [dbo].[CRE_SIGTasaPromedioPonderadaCLI_SP]
	@dFechaSel DATETIME, 
	@idAgencia INT, 
	@idZona INT, 
	@idUsuario INT
AS
SET NOCOUNT ON;
BEGIN
	DECLARE @dFechaIniMesAct DATE,
			@dFechaFinMesAnt DATE,
			@dFechaIniMesAnt DATE,
			@nTipoCambioMesAnt DECIMAL(14,4),
			@nTipoCambioMesAct DECIMAL(14,4),
			@nTipoCambioFecSel DECIMAL(14,4)

	SET @dFechaIniMesAct = DATEFROMPARTS(YEAR(@dFechaSel), MONTH(@dFechaSel), 1)
	SET @dFechaFinMesAnt = DATEADD(DAY,-1,@dFechaIniMesAct)
	SET @dFechaIniMesAnt = DATEFROMPARTS(YEAR(@dFechaFinMesAnt), MONTH(@dFechaFinMesAnt), 1)

	DECLARE @dCierrMesAnt VARCHAR(20),
			@dFechaActVar DATE,
			@dMesActVar VARCHAR(20)
	SET @dCierrMesAnt = UPPER(CAST(FORMAT(@dFechaFinMesAnt, 'Y', 'es-PE') AS VARCHAR(100)))
	SET @dFechaActVar = @dFechaSel
	SET @dMesActVar = UPPER(CAST(FORMAT(@dFechaSel, 'Y', 'es-PE') AS VARCHAR(100)))

	-- ======================================================================================
	-- Obteniendo tipo de cambio al cierre del mes anterior y de la fecha actual
	-- ======================================================================================

	SELECT @nTipoCambioMesAnt = nTipCamFij FROM SI_FinTipoCambio
	WHERE dFecTipCambio = @dFechaFinMesAnt

	SELECT @nTipoCambioMesAct = nTipCamFij FROM SI_FinTipoCambio
	WHERE dFecTipCambio = @dFechaIniMesAct

	SELECT @nTipoCambioFecSel = nTipCamFij FROM SI_FinTipoCambio
	WHERE dFecTipCambio = @dFechaSel

	-- ======================================================================================
	-- Tasa promedio ponderada INI
	-- ======================================================================================

	DECLARE @tTPPIni TABLE(
		idAgencia INT, idUsuario INT, nTasa DECIMAL(20,4), idEstablecimiento INT
	)

	-- ======================================================================================
	-- Tasa promedio ponderada Cierre
	-- ======================================================================================

	DECLARE @tTPPCierre TABLE(
		idAgencia INT, idUsuario INT, nTasa DECIMAL(20,4), idEstablecimiento INT
	)

	-- ======================================================================================
	-- Tasa promedio ponderada Actual
	-- ======================================================================================

	DECLARE @tTPPActual TABLE(
		idAgencia INT, idUsuario INT, nTasa DECIMAL(20,4), idEstablecimiento INT
	)

	-- ======================================================================================
	-- Tasa promedio ponderada Cierre
	-- ======================================================================================

	DECLARE @tTPPDesembolsoCierre TABLE(
		idAgencia INT, idUsuario INT, nTasa DECIMAL(20,4), idEstablecimiento INT
	)

	-- ======================================================================================
	-- Tasa promedio ponderada INI
	-- ======================================================================================

	DECLARE @tTPPDesembolsoIni TABLE(
		idAgencia INT, idUsuario INT, nTasa DECIMAL(20,4), idEstablecimiento INT
	)

	-- ======================================================================================
	-- Desembolsos inicio de mes actual hasta fecha actual
	-- ======================================================================================

	DECLARE @tSDesembolsosAct TABLE(
		idAgencia INT, idUsuario INT, nCapitalDesembolso DECIMAL(20,4), nOperaciones INT, idEstablecimiento INT
	)

	-- ======================================================================================
	-- Saldo Capital cierre del mes anterior
	-- ======================================================================================

	DECLARE @tSCMAnt TABLE(
		idAgencia INT, idUsuario INT, nSaldo DECIMAL(20,4), idEstablecimiento INT
	)

	-- ======================================================================================
	-- Saldo Capital al inicio del mes actual
	-- ======================================================================================

	DECLARE @tSCIniMActual TABLE(
		idAgencia INT, idUsuario INT, nSaldo DECIMAL(20,4), idEstablecimiento INT
	)

	-- ======================================================================================
	-- Saldo Capital a la fecha
	-- ======================================================================================

	DECLARE @tSCFActual TABLE(
		idAgencia INT, idUsuario INT, nSaldo DECIMAL(20,4), idEstablecimiento INT
	)

	-- ======================================================================================
	-- Tasa promedio ponderada Actual
	-- ======================================================================================

	DECLARE @tTPPDesembolsoActual TABLE(
		idAgencia INT, idUsuario INT, nTasa DECIMAL(20,4), idEstablecimiento INT
	)

	-- ======================================================================================
	-- INSERTANDO CREDITOS VENCIDOS DURANTE EL PERIODO DEL MES DE LA FECHA SELECCIONADA
	-- ======================================================================================

	SELECT idCuenta
	into #tCuentasCancPerVige
	FROM SI_FinSaldoCreditos
	WHERE idTipoCuentaCredito = 1
		AND idEstado = 5
		AND dFecha = @dFechaFinMesAnt
	GROUP BY idCuenta
	
	SELECT A.idCuenta, B.idAgencia, B.idUsuario, B.idEstablecimiento
	INTO #CreditosTempIni 
	FROM #tCuentasCancPerVige A
		LEFT JOIN SI_FinSaldoCreditos B
			ON A.idCuenta = B.idCuenta
			AND B.dFecha = @dFechaSel
	GROUP BY A.idCuenta, B.idAgencia, B.idUsuario, B.idEstablecimiento

	UPDATE #CreditosTempIni SET
		idAgencia = A.idAgencia,
		idUsuario = A.idUsuario,
		idEstablecimiento = A.idEstablecimiento
	FROM SI_FinCreditos A
		INNER JOIN #CreditosTempIni B
			ON A.idCuenta = B.idCuenta
				AND B.idUsuario IS NULL
				AND B.idAgencia IS NULL
				AND B.idEstablecimiento IS NULL
	
	SELECT idAgencia			, idUsuario				, IdMoneda				, nSaldo
		, idEstablecimiento 	, nAtraso = MAX(nAtraso) OVER (PARTITION BY idcuenta)
		, idCondicionContable	, dFechaDesembolso
		, nTasaCompensatoria	, idCli					, idCuenta				, dFecha
	INTO #tSaldoCredAct
	FROM SI_FinSaldoCreditos 
	WHERE dFecha = @dFechaSel
		AND idEstado = 5
		AND idTipoCuentaCredito = 1

	SELECT idAgencia			, idUsuario				, IdMoneda				, nSaldo
		, idEstablecimiento 	, nAtraso= MAX(nAtraso)	OVER (PARTITION BY idcuenta )
		, idCondicionContable	, dFechaDesembolso
		, nTasaCompensatoria	, idCli					, idCuenta				, dFecha
	INTO #tSaldoCredCierAnt
	FROM SI_FinSaldoCreditos 
	WHERE dFecha = @dFechaFinMesAnt
		AND idEstado = 5
		AND idTipoCuentaCredito = 1

	SELECT idAgencia	
		, nTasa = IIF(SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel)) = 0, 0,(
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel ) * A.nTasaCompensatoria/100.0000 )
					/
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel))
					))
	INTO #tTPPAgenciaCierre
	FROM #tSaldoCredCierAnt A
	GROUP BY A.idAgencia

	SELECT idAgencia			
		, nTasa = IIF(SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel)) = 0, 0,(
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel ) * A.nTasaCompensatoria/100.0000 )
					/
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel))
					))
	INTO #tTPPAgenciaActual
	FROM #tSaldoCredAct A	
	GROUP BY A.idAgencia

	SELECT c.idZona	
		, nTasa = IIF(SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel)) = 0, 0,(
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel ) * A.nTasaCompensatoria/100.0000 )
					/
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel))
					))
	INTO #tTPPZonaCierre
	FROM #tSaldoCredCierAnt A
		INNER JOIN SI_FinZonaAgencia C
			ON A.idAgencia = C.idAgencia
			AND C.lVigente = 1
	GROUP BY C.idZona

	SELECT c.idZona			
		, nTasa = IIF(SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel)) = 0, 0,(
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel ) * A.nTasaCompensatoria/100.0000 )
					/
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel))
					))
	INTO #tTPPZonaActual
	FROM #tSaldoCredAct A	
		INNER JOIN SI_FinZonaAgencia C
			ON A.idAgencia = C.idAgencia
			AND C.lVigente = 1
	GROUP BY C.idZona

	SELECT a.idAgencia
		, nTasa = IIF(SUM(IIF(B.IdMoneda = 1, B.nSaldo , B.nSaldo * @nTipoCambioFecSel)) = 0, 0,(
					SUM(IIF(B.IdMoneda = 1, B.nSaldo , B.nSaldo * @nTipoCambioFecSel ) * B.nTasaCompensatoria/100.0000 )
					/
					SUM(IIF(B.IdMoneda = 1, B.nSaldo , B.nSaldo * @nTipoCambioFecSel))
					))
	INTO #tTPPAgenciaINI
	FROM #CreditosTempIni A
		INNER JOIN SI_FinSaldoCreditos B
			ON A.idCuenta = B.idCuenta
	WHERE B.idTipoCuentaCredito = 1
		AND B.dFecha = @dFechaFinMesAnt
	GROUP BY A.idAgencia

	SELECT C.idZona							
		, nTasa = IIF(SUM(IIF(B.IdMoneda = 1, B.nSaldo , B.nSaldo * @nTipoCambioFecSel)) = 0, 0,(
					SUM(IIF(B.IdMoneda = 1, B.nSaldo , B.nSaldo * @nTipoCambioFecSel ) * B.nTasaCompensatoria/100.0000 )
					/
					SUM(IIF(B.IdMoneda = 1, B.nSaldo , B.nSaldo * @nTipoCambioFecSel))
					))
	INTO #tTPPZonaINI
	FROM #CreditosTempIni A
		INNER JOIN SI_FinSaldoCreditos B
			ON A.idCuenta = B.idCuenta
		INNER JOIN SI_FinZonaAgencia C
			ON A.idAgencia = C.idAgencia
			AND C.lVigente = 1
	WHERE B.idTipoCuentaCredito = 1
		AND B.dFecha = @dFechaFinMesAnt
	GROUP BY C.idZona

	SELECT idAgencia
		, nTasa = IIF(SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel)) = 0, 0,(
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel ) * A.nTasaCompensatoria/100.0000 )
					/
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel))
					))
	INTO #tTPPDesAgenciaActual
	FROM #tSaldoCredAct A
	WHERE A.dFechaDesembolso BETWEEN @dFechaIniMesAct AND @dFechaSel
	GROUP BY A.idAgencia

	SELECT C.idZona
		, nTasa = IIF(SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel)) = 0, 0,(
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel ) * A.nTasaCompensatoria/100.0000 )
					/
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel))
					))
	INTO #tTPPDesZonaActual
	FROM #tSaldoCredAct A
	INNER JOIN SI_FinZonaAgencia C
			ON A.idAgencia = C.idAgencia
			AND C.lVigente = 1
	WHERE A.dFechaDesembolso BETWEEN @dFechaIniMesAct AND @dFechaSel
	GROUP BY C.idZona

	SELECT a.idAgencia
		, nTasa = IIF(SUM(IIF(B.IdMoneda = 1, B.nSaldo , B.nSaldo * @nTipoCambioFecSel)) = 0, 0,(
					SUM(IIF(B.IdMoneda = 1, B.nSaldo , B.nSaldo * @nTipoCambioFecSel ) * B.nTasaCompensatoria/100.0000 )
					/
					SUM(IIF(B.IdMoneda = 1, B.nSaldo , B.nSaldo * @nTipoCambioFecSel))
					))
	INTO #tTPPDesAgenciaIni
	FROM #CreditosTempIni A
		INNER JOIN SI_FinSaldoCreditos B
			ON A.idCuenta = B.idCuenta
	WHERE B.idTipoCuentaCredito = 1
		AND dFecha = @dFechaFinMesAnt
		AND dFechaDesembolso BETWEEN @dFechaIniMesAnt AND @dFechaFinMesAnt
	GROUP BY A.idAgencia

	SELECT C.idZona
		, nTasa = IIF(SUM(IIF(B.IdMoneda = 1, B.nSaldo , B.nSaldo * @nTipoCambioFecSel)) = 0, 0,(
					SUM(IIF(B.IdMoneda = 1, B.nSaldo , B.nSaldo * @nTipoCambioFecSel ) * B.nTasaCompensatoria/100.0000 )
					/
					SUM(IIF(B.IdMoneda = 1, B.nSaldo , B.nSaldo * @nTipoCambioFecSel))
					))
	INTO #tTPPDesZonaIni
	FROM #CreditosTempIni A
		INNER JOIN SI_FinSaldoCreditos B
			ON A.idCuenta = B.idCuenta
		INNER JOIN SI_FinZonaAgencia C
			ON A.idAgencia = C.idAgencia
			AND C.lVigente = 1
	WHERE B.idTipoCuentaCredito = 1
		AND dFecha = @dFechaFinMesAnt
		AND dFechaDesembolso BETWEEN @dFechaIniMesAnt AND @dFechaFinMesAnt
	GROUP BY C.idZona

	SELECT A.idAgencia
		, nTasa = IIF(SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel)) = 0, 0,(
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel ) * A.nTasaCompensatoria/100.0000 )
					/
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel))
					))
	INTO #tTPPDesAgenciaCierre
	FROM #tSaldoCredCierAnt A
	WHERE dFechaDesembolso BETWEEN @dFechaIniMesAnt AND @dFechaFinMesAnt
	GROUP BY A.idAgencia;

	SELECT c.idZona
		, nTasa = IIF(SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel)) = 0, 0,(
					SUM(IIF(A.IdMoneda = 1, A.nSaldo , A.nSaldo * @nTipoCambioFecSel ) * A.nTasaCompensatoria/100.0000 )
					/

	WHERE A.idAgencia = IIF(@idAgencia = 0, A.idAgencia , @idAgencia)

CREATE PROCEDURE GEN_GuardarExpediente_SP
	 @cXmlExpediente XML,
	 @idSolicitud INT,
	 @idUsuario INT,
	 @cTipoSolicitud VARCHAR(20),
	 @cGrupo VARCHAR(120)
AS
SET NOCOUNT ON

BEGIN TRY	

	DECLARE @dFechaAct DATE, @idEstado INT, @cDetalleArchivo VARCHAR(50), @lAtendido BIT

	SELECT @dFechaAct = cValVar FROM dbo.SI_FinVariable WHERE cVariable = 'dFechaAct' AND idAgencia = 1 AND lVigente = 1

	DECLARE @idExpediente INT, @idCli INT = NULL
	
	IF(@cTipoSolicitud = 'individual')
	BEGIN
		SELECT @idCli = idCli FROM dbo.SI_FinSolicitud WHERE idSolicitud = @idSolicitud    
	END
	
	IF EXISTS(SELECT 1 FROM dbo.SI_FinSolicitud WHERE idSolicitud = @idSolicitud AND idEstado = 19) --DEVUELTO
	BEGIN		
		SET @cDetalleArchivo = 'Devuelto'
	END

	EXEC SP_XML_PREPAREDOCUMENT @idExpediente OUTPUT, @cXmlExpediente

		
	SELECT idExpediente, cXmlExpediente, idDetGrupo, idSecundario
	INTO #tmpExpediente
	FROM  OpenXml (@idExpediente,N'/dsExpedienteDatos/dtExpediente',2)
	WITH (idExpediente INT, cXmlExpediente XML, idDetGrupo INT, idSecundario INT);

	EXEC SP_XML_REMOVEDOCUMENT @idExpediente;

	DECLARE @idTipoArchivo INT, @idFormato INT, @idCategoriaArchivo INT
	SELECT @idTipoArchivo = idTipoArchivo  FROM dbo.SI_FINTipoArchivo WHERE cExtencion = '.rdlc' AND cNombre = 'Local Report ' AND lVigente = 1
	SELECT @idFormato = idFormato FROM dbo.SI_FinFormatoArchivo WHERE cFormato = 'xml' AND lVigente = 1
	SELECT @idCategoriaArchivo = idCategoriaArchivo FROM dbo.SI_FINCategoriaArchivo WHERE cNombreCategoria = 'Expediente' AND lVigente = 1

	DECLARE @idDescPosIntral INT, @idExpedPosInt INT
	SELECT @idExpedPosInt = idExpediente FROM dbo.SI_FinExpediente WHERE cStoreProcedure = 'CRE_RptPosicionIntegralIntervinientes_SP' AND lVigente = 1
	SELECT @idDescPosIntral = idDescTipoDoc FROM dbo.SI_FINDescTipoDoc WHERE cDescTipoDoc = 'Posición Integral de Intervinientes' AND lVigente = 1
	

	SELECT
		cNombreArchivo = C.cExpediente, idTipoArchivo = @idTipoArchivo, cArchivo = NULL, @cDetalleArchivo cDetalleArchivo, idCli = IIF(@cTipoSolicitud ='individual',A.idSecundario,NULL),
		 idVisita = NULL, idSolicitud = IIF(@cTipoSolicitud = 'individual',@idSolicitud,NULL), idModulo = 1, idCategoriaArchivo = @idCategoriaArchivo,
		idUsuReg = @idUsuario, dFechaReg = @dFechaAct, idUsuMod = NULL, dFechaMod = NULL, idDescTipoDoc = IIF(A.idExpediente = @idExpedPosInt, @idDescPosIntral, B.idDescTipoDoc),
		 lVigente = 1, idDetalleVisita = NULL, idSolicitudCredGrupoSol = IIF(@cTipoSolicitud = 'individual',NULL, @idSolicitud),
		idArchivoForeing = A.idExpediente, idFormato = @idFormato, cArchivoXml = CONVERT(XML, A.cXmlExpediente.value('.', 'varchar(max)')), bArchivoBinary = NULL,
		C.lHistorico, A.idSecundario
	INTO #tmpExpedienteB
	FROM #tmpExpediente A
	INNER JOIN dbo.SI_FinDetGrupoExpediente B
		ON A.idDetGrupo = B.idDetgrupo
	INNER JOIN dbo.SI_FinExpediente C
		ON A.idExpediente = C.idExpediente
		AND C.lVigente = 1

	/*Si pertenece a Grupo Solidario*/
	DECLARE @idSolGrupoSolidario INT = NULL
	IF(@cTipoSolicitud = 'grupal')
	BEGIN
	    SET @idSolGrupoSolidario = @idSolicitud
	END

	SELECT @idSolGrupoSolidario = idSolicitudCredGrupoSol FROM dbo.SI_FinSolicitud WHERE idSolicitud = @idSolicitud AND @cTipoSolicitud = 'individual'

	IF EXISTS(SELECT 1
				FROM dbo.SI_FinArchivo A
				INNER JOIN #tmpExpedienteB B
					ON A.idArchivoForeing = B.idArchivoForeing
					AND A.idCategoriaArchivo = 2
					--AND A.lVigente = 1
					AND (A.idDescTipoDoc = 24 OR A.lVigente = 1)
					AND (B.lHistorico = 0 OR B.lHistorico IS NULL)
					AND @cTipoSolicitud <> 'individual'
					AND A.idSolicitudCredGrupoSol = @idSolicitud
					AND ISNULL(A.idCli,0) = ISNULL(B.idSecundario,0)
					AND A.idDescTipoDoc = B.idDescTipoDoc
					UNION
					SELECT 1
					FROM dbo.SI_FinArchivo A
					INNER JOIN #tmpExpedienteB B
						ON A.idArchivoForeing = B.idArchivoForeing
						AND A.idCategoriaArchivo = 2
						--AND A.lVigente = 1
						AND (A.idDescTipoDoc = 24 OR A.lVigente = 1)
						AND (B.lHistorico = 0 OR B.lHistorico IS NULL)
						AND @cTipoSolicitud = 'individual'
						AND A.idSolicitud = @idSolicitud
						AND ISNULL(A.idCli,0) = ISNULL(B.idSecundario,0)
						AND A.idDescTipoDoc = B.idDescTipoDoc
						)
						
		BEGIN
		    UPDATE A
				SET A.cNombreArchivo = B.cNombreArchivo, A.cArchivoXml = B.cArchivoXml, A.idUsuMod = @idUsuario, A.dFechaMod = GETDATE()
				, A.idSolicitudCredGrupoSol = @idSolGrupoSolidario, A.lVigente = 1
			FROM dbo.SI_FinArchivo A
			INNER JOIN #tmpExpedienteB B
				ON A.idArchivoForeing = B.idArchivoForeing
				AND A.idCategoriaArchivo = 2

	          B.idArchivoForeing ,B.idFormato ,B.cArchivoXml ,B.bArchivoBinary
			  FROM #tmpExpedienteB B
        END

	DROP TABLE #tmpExpediente
	DROP TABLE #tmpExpedienteB
    
	SELECT 1 AS nRespuesta, 'Los Expedientes de este Proceso se Guardaron Correctamente.' AS cRespuesta
END TRY
BEGIN CATCH

	IF(@cGrupo = 'Solicitud de Credito')

				nAnalisisHoriz	decimal(14, 4)	'nAnalisisHoriz'	,
				nTotalMN		decimal(14, 2)	'nTotalMN'			,
				nTotalME		decimal(14, 2)	'nTotalME'	
			) AS b
			WHERE SI_FinEstResEval.idEstResEval = b.idEstResEval;
			
			EXEC sp_xml_removedocument @hdocEstResultados;

			-- =============================================================================================================
			-- DB [dbo].[SI_FinDestCredProp]: UPDATE, INSERT Y DELETE
			-- =============================================================================================================
			DECLARE @tblDestCredProp TABLE (
				idDestCredProp  INT PRIMARY KEY,
				idEvalCred      INT,
				idTipDestCred   INT,
				idDetalle       INT,
				nPorcentaje     DECIMAL(14, 2),
				nMonto          DECIMAL(14, 2)
			);

			INSERT INTO @tblDestCredProp
			SELECT idDestCredProp, idEvalCred, idTipDestCred, idDetalle, nPorcentaje, nMonto
			FROM OPENXML(@hdocDestCredProp, '/DestCredProp/Item')
			WITH 
			(
				idDestCredProp  INT             'idDestCredProp',
				idEvalCred      INT             'idEvalCred',
				idTipDestCred   INT             'idTipDestCred',
				idDetalle       INT             'idDetalle',
				nPorcentaje     DECIMAL(14, 2)  'nPorcentaje',
				nMonto          DECIMAL(14, 2)  'nMonto'                
			);
			EXEC sp_xml_removedocument @hdocDestCredProp;
			
			DELETE T
			FROM dbo.SI_FinDestCredProp T
			LEFT JOIN @tblDestCredProp S ON T.idDestCredProp = S.idDestCredProp
			WHERE T.idEvalCred = @idEvalCred 
			AND S.idDestCredProp IS NULL;

			UPDATE T
			SET 
				T.idTipDestCred = S.idTipDestCred, 
				T.idDetalle     = S.idDetalle, 
				T.nPorcentaje   = S.nPorcentaje, 
				T.nMonto        = S.nMonto
			FROM dbo.SI_FinDestCredProp T
			INNER JOIN @tblDestCredProp S ON T.idDestCredProp = S.idDestCredProp
			WHERE 
				T.idTipDestCred <> S.idTipDestCred OR 
				T.idDetalle     <> S.idDetalle     OR 
				T.nPorcentaje   <> S.nPorcentaje   OR 
				T.nMonto        <> S.nMonto;

			INSERT INTO dbo.SI_FinDestCredProp (idEvalCred, idTipDestCred, idDetalle, nPorcentaje, nMonto, lVigente)
			SELECT @idEvalCred, S.idTipDestCred, S.idDetalle, S.nPorcentaje, S.nMonto, 1
			FROM @tblDestCredProp AS S
			WHERE NOT EXISTS (
				SELECT 1 
				FROM dbo.SI_FinDestCredProp T 
				WHERE T.idDestCredProp = S.idDestCredProp
			);

			-- =============================================================================================================
			-- DB [dbo].[SI_FINEvalRcc]: UPDATE, INSERT
			-- =============================================================================================================
			UPDATE SI_FINEvalRcc SET lVigente = 0
			WHERE idEvalCre = @idEvalCred

			UPDATE T
			SET 
				T.nMontoMonOrig = S.nMontoMonOrig,
				T.nPrimer       = CONVERT(DECIMAL(14,4), S.nPrimer),
				T.nSegundo      = S.nSegundo,
				T.nTercero      = S.nTercero,
				T.lVigente      = 1
			FROM dbo.SI_FINEvalRcc AS T
			INNER JOIN @EvalRcc AS S ON T.idEvalCre = S.idEvalCre AND T.nOrden = S.nOrden
			WHERE 
				T.nMontoMonOrig <> S.nMontoMonOrig OR
				T.nPrimer       <> CONVERT(DECIMAL(14,4), S.nPrimer) OR
				T.nSegundo      <> S.nSegundo OR
				T.nTercero      <> S.nTercero

			INSERT INTO dbo.SI_FINEvalRcc 
				(idOpe, idEvalCre, nTipDato, cDescripcion, idMoneda, nMontoMonOrig, nPrimer, nSegundo, nTercero, lVigente, nOrden)
			SELECT 
				S.idOpe, S.idEvalCre, S.nTipDato, S.cDescripcion, S.idMoneda, S.nMontoMonOrig, 
				CONVERT(DECIMAL(14,4), S.nPrimer), S.nSegundo, S.nTercero, 1, S.nOrden
			FROM @EvalRcc AS S
			WHERE NOT EXISTS (
				SELECT 1 
				FROM dbo.SI_FINEvalRcc AS T 
				WHERE T.idEvalCre = S.idEvalCre 
				AND T.nOrden = S.nOrden
			);
			-- =============================================================================================================
			-- DB [dbo].[SI_FinIndicadorEval]: UPDATE indicadores financieros
			-- =============================================================================================================
			UPDATE SI_FinIndicadorEval SET 
				nRatio	= b.nRatio	
			FROM OPENXML(@hdocIndicadorEval, '/IndicadorEval/Item')
			WITH 
			(
				idIndicadorEval		INT				'idIndicadorEval',
				nRatio				decimal(14, 8)	'nRatio'		
			) AS b
			WHERE SI_FinIndicadorEval.idIndicadorEval = b.idIndicadorEval;
			
			EXEC sp_xml_removedocument @hdocIndicadorEval;

		COMMIT TRANSACTION
	END TRY
	BEGIN CATCH
		DECLARE @cErrorMessage	NVARCHAR(4000) = ERROR_MESSAGE(),
				@nErrorSeverity	INT = ERROR_SEVERITY(),
				@nErrorState	INT = ERROR_STATE()

		ROLLBACK TRANSACTION
		RAISERROR(@cErrorMessage, @nErrorSeverity, @nErrorState) WITH NOWAIT
		RETURN
	END CATCH

	SELECT '0' as idMsje , 'Evaluación Guardada Correctamente.' as cMsje, @idEvalCred as idEvalCred
END
