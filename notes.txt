CREATE PROCEDURE [dbo].[CRE_ValidacionMontoCampana_SP]
    @idCliente      INT,
    @idProducto     INT,
    @idOperacion    INT
AS
SET NOCOUNT ON
BEGIN
    DECLARE @cListaCampanaExclusionMonto VARCHAR(1000),
            @cOpeExclusionMontoCampana VARCHAR(1000);

    SELECT @cListaCampanaExclusionMonto = cValVar
    FROM dbo.SI_FinVariable
    WHERE cVariable = 'cCampanaExclusionMonto' AND lVigente = 1;

    SELECT @cOpeExclusionMontoCampana = cValVar
    FROM dbo.SI_FinVariable
    WHERE cVariable = 'cOpeExclusionMontoCampana' AND lVigente = 1;

    IF EXISTS (
        SELECT 1
        FROM dbo.GEN_ConverCadenaTabla_Fn(@cOpeExclusionMontoCampana)
        WHERE CAST(cValor AS INT) = @idOperacion
    )
    BEGIN
        SELECT TOP (0)
            0 AS idProducto,
            0 AS nMontoMaximoPermitido,
            0 AS idGrupoCamp,
            CAST(0 AS BIT) AS lPreAprobado,
            '' AS cGrupoCamp,
            CAST(0 AS BIT) AS lCampanaPropuesta,
            '' AS cTipoGrupoCamp;
        RETURN;
    END

    DECLARE @exclusiones TABLE (idGrupoCamp INT PRIMARY KEY);
    INSERT INTO @exclusiones (idGrupoCamp)
    SELECT CAST(cValor AS INT)
    FROM dbo.GEN_ConverCadenaTabla_Fn(@cListaCampanaExclusionMonto);

      SELECT
          A.idProducto,
          ISNULL(D.nMonto, 0) AS nMontoMaximoPermitido,
          A.idGrupoCamp,
          CAST(ISNULL(A.lPreAprobado, 0) AS BIT) AS lPreAprobado,
          B.cGrupoCamp,
          CAST(IIF(D.idDetalle IS NULL, 0, 1) AS BIT) AS lCampanaPropuesta,
          ISNULL(E.cDescripcion, '') AS cTipoGrupoCamp,
          D.idTipoGrupoCamp
      FROM dbo.SI_FinGrupoCampProducto A WITH (NOLOCK)
      INNER JOIN dbo.SI_FinGrupoCamp B WITH (NOLOCK)
          ON B.idGrupoCamp = A.idGrupoCamp AND B.lVigente = 1
      INNER JOIN dbo.SI_FinOpeGrupoCamp C WITH (NOLOCK)
          ON C.idGrupoCamp = A.idGrupoCamp AND C.idOperacion = @idOperacion
      LEFT JOIN (
          SELECT nMonto, idTipoGrupoCamp, idGrupoCamp, idDetalle
          FROM dbo.SI_FinDetallePreAprobados
          WHERE idCli = @idCliente AND lVigente = 1 AND nMonto <> 0
      ) D ON D.idGrupoCamp = A.idGrupoCamp
      LEFT JOIN dbo.SI_FinTipoGrupoCamp E WITH (NOLOCK)
          ON E.idTipoGrupoCamp = D.idTipoGrupoCamp AND E.lVigente = 1
      WHERE A.idProducto = @idProducto
        AND A.lVigente = 1
        AND NOT EXISTS (SELECT 1 FROM @exclusiones F WHERE F.idGrupoCamp = A.idGrupoCamp)
      ORDER BY D.nMonto DESC;
END
GO
GRANT EXECUTE ON [dbo].[CRE_ValidacionMontoCampana_SP] TO [Rol_Fin]
GO
